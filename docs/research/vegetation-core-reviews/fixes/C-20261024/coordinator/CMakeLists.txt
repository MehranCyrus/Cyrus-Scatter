cmake_minimum_required(VERSION 3.20)
project(CyrusVegetationABCReview LANGUAGES CXX)
include(CTest)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../abc-starter" core)
add_executable(cyrus_abc_review pipeline.cpp)
target_compile_features(cyrus_abc_review PRIVATE cxx_std_20)
target_link_libraries(cyrus_abc_review PRIVATE cyrus_vegetation_collision cyrus_vegetation_fitter cyrus_vegetation_delivery)
foreach(case IN ITEMS catalog stored_candidate clear_rest unfit_rest corrupt_mesh root_conflict two_spheres rotated_boxes_conservative_rejection rotated_boxes_guard_band clamp_frame)
    add_test(NAME "pipeline_${case}" COMMAND cyrus_abc_review "${case}")
    set_tests_properties("pipeline_${case}" PROPERTIES TIMEOUT 30)
endforeach()
foreach(case IN ITEMS invalid_root_mode nonfinite_inactive small_angle frame_budget)
    add_test(NAME "review_${case}" COMMAND cyrus_abc_review "${case}")
    set_tests_properties("review_${case}" PROPERTIES TIMEOUT 30)
endforeach()
