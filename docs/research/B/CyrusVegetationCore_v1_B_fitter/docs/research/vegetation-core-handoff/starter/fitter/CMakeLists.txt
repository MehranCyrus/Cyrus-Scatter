if(NOT TARGET cyrus_prototype_contract)
  message(FATAL_ERROR "Configure the frozen starter scaffold; cyrus_prototype_contract is required.")
endif()
add_library(cyrus_vegetation_fitter STATIC fitter.cpp)
target_link_libraries(cyrus_vegetation_fitter PUBLIC cyrus_prototype_contract)
target_compile_features(cyrus_vegetation_fitter PUBLIC cxx_std_20)
option(CYRUS_FITTER_SANITIZERS "Enable address and undefined-behavior sanitizers for B's targets" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cyrus_vegetation_fitter PRIVATE -Wall -Wextra -Wpedantic)
  if(CYRUS_FITTER_SANITIZERS)
    target_compile_options(cyrus_vegetation_fitter PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cyrus_vegetation_fitter PUBLIC -fsanitize=address,undefined)
  endif()
elseif(MSVC)
  target_compile_options(cyrus_vegetation_fitter PRIVATE /W4 /permissive-)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
