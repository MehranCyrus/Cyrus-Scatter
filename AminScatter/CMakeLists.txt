cmake_minimum_required(VERSION 3.24)
project(AminScatter VERSION 0.1.0 LANGUAGES CXX)
option(AMIN_BUILD_MAX "Build the 3ds Max extension" ON)
if(AMIN_BUILD_MAX)
  include("${CMAKE_CURRENT_LIST_DIR}/../cmake/CyrusMaxSDK.cmake")
endif()
find_package(Threads REQUIRED)
add_library(amin_scatter STATIC src/scatter.cpp src/execution.cpp src/brush.cpp)
target_link_libraries(amin_scatter PUBLIC Threads::Threads)
target_include_directories(amin_scatter PUBLIC include)
target_compile_features(amin_scatter PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(amin_scatter PRIVATE /W4 /WX /permissive-)
endif()
enable_testing()
add_executable(brush_tests tests/brush_tests.cpp)
target_link_libraries(brush_tests PRIVATE amin_scatter)
add_test(NAME scatter_brush COMMAND brush_tests)
add_test(NAME scatter_brush_pick_fixture COMMAND brush_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/brush_sphere_pole.txt")
option(AMIN_BUILD_BRUSH_LAB "Build the isolated procedural Brush prototype" OFF)
option(AMIN_BUILD_BRUSH_BENCHMARK "Build the isolated Brush history cost probe" OFF)
if(AMIN_BUILD_BRUSH_BENCHMARK)
  add_executable(brush_history_benchmark ../tools/brush_lab/history_benchmark.cpp)
  target_link_libraries(brush_history_benchmark PRIVATE amin_scatter)
endif()
add_executable(preview_sampling_tests tests/preview_sampling_tests.cpp)
target_link_libraries(preview_sampling_tests PRIVATE amin_scatter)
add_test(NAME scatter_preview_sampling COMMAND preview_sampling_tests)
add_executable(threading_tests tests/threading_tests.cpp)
target_link_libraries(threading_tests PRIVATE amin_scatter)
add_test(NAME scatter_threading COMMAND threading_tests)
add_executable(prepared_band_tests tests/prepared_band_tests.cpp)
target_link_libraries(prepared_band_tests PRIVATE amin_scatter)
add_test(NAME scatter_prepared_bands COMMAND prepared_band_tests)
option(AMIN_BUILD_BENCHMARKS "Build the controlled CPU benchmark" OFF)
if(AMIN_BUILD_BENCHMARKS)
  add_executable(scatter_benchmark tests/compute_benchmark.cpp)
  target_compile_definitions(scatter_benchmark PRIVATE CYRUS_BENCH_EXECUTION=1)
  target_link_libraries(scatter_benchmark PRIVATE amin_scatter)
endif()
add_executable(scatter_tests tests/scatter_tests.cpp)
target_link_libraries(scatter_tests PRIVATE amin_scatter)
add_test(NAME scatter_core COMMAND scatter_tests)
add_executable(spacing_tests tests/spacing_tests.cpp)
target_link_libraries(spacing_tests PRIVATE amin_scatter)
add_test(NAME scatter_spacing COMMAND spacing_tests)
add_executable(weight_tests tests/weight_tests.cpp)
target_link_libraries(weight_tests PRIVATE amin_scatter)
add_test(NAME scatter_weights COMMAND weight_tests)
add_executable(orientation_tests tests/orientation_tests.cpp)
target_link_libraries(orientation_tests PRIVATE amin_scatter)
add_test(NAME scatter_orientation COMMAND orientation_tests)
add_executable(edge_border_tests tests/edge_border_tests.cpp)
target_link_libraries(edge_border_tests PRIVATE amin_scatter)
add_test(NAME scatter_edge_border COMMAND edge_border_tests)
add_executable(boundary_falloff_tests tests/boundary_falloff_tests.cpp)
target_link_libraries(boundary_falloff_tests PRIVATE amin_scatter)
add_test(NAME scatter_boundary_falloff COMMAND boundary_falloff_tests)
if(AMIN_BUILD_MAX)
  if(AMIN_BUILD_BRUSH_LAB)
    add_library(CyrusBrushLab MODULE src/brush_lab.cpp)
    target_link_libraries(CyrusBrushLab PRIVATE amin_scatter core geom mesh gfx maxutil Maxscrpt paramblk2 ole32)
    cyrus_max_target(CyrusBrushLab)
    set_target_properties(CyrusBrushLab PROPERTIES SUFFIX ".dlx")
    # DLX loads scripting primitives, but cannot register saved scene classes.
    add_library(CyrusBrushStorage MODULE src/brush_storage_plugin.cpp)
    add_dependencies(CyrusBrushStorage CyrusBrushLab)
    target_link_libraries(CyrusBrushStorage PRIVATE "$<TARGET_FILE_DIR:CyrusBrushLab>/CyrusBrushLab.lib")
    cyrus_max_target(CyrusBrushStorage)
    set_target_properties(CyrusBrushStorage PROPERTIES SUFFIX ".dlh")
  endif()
  add_library(AminScatter MODULE src/max_bridge.cpp src/preview.cpp src/point_display.cpp src/cyrus_edit.cpp src/cyrus_edit.rc)
  target_link_libraries(AminScatter PRIVATE amin_scatter core geom mesh gfx maxutil Maxscrpt paramblk2 ole32 GraphicsDriver GraphicsUtility DataBridge)
  cyrus_max_target(AminScatter)
  set_target_properties(AminScatter PROPERTIES SUFFIX ".dlx")
  add_library(CyrusScatterEdit MODULE src/edit_plugin.cpp)
  target_compile_features(CyrusScatterEdit PRIVATE cxx_std_17)
  add_dependencies(CyrusScatterEdit AminScatter)
  # MODULE targets expose their DLL as TARGET_LINKER_FILE; explicitly use the
  # generated MSVC import library, in the correct single/multi-config directory.
  target_link_libraries(CyrusScatterEdit PRIVATE "$<TARGET_FILE_DIR:AminScatter>/AminScatter.lib")
  cyrus_max_target(CyrusScatterEdit)
  set_target_properties(CyrusScatterEdit PROPERTIES SUFFIX ".dlm")
endif()
