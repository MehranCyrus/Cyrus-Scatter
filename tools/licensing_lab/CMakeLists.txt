cmake_minimum_required(VERSION 3.24)
project(CyrusLicenseLab LANGUAGES CXX)
option(CYRUS_BUILD_LICENSING_LAB "Build UNSIGNED SYNTHETIC licensing experiment" OFF)
if(NOT CYRUS_BUILD_LICENSING_LAB)
  message(FATAL_ERROR "This test-only target requires CYRUS_BUILD_LICENSING_LAB=ON; never package it")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/CyrusMaxSDK.cmake")
option(CYRUS_BUILD_SIGNED_LICENSE_LAB "Build ephemeral-issuer signature experiment" OFF)
option(CYRUS_BUILD_LOCAL_CLIENT_LAB "Build private installation/activation client, NEVER SHIP" OFF)
if(CYRUS_BUILD_LOCAL_CLIENT_LAB)
  if(NOT EXISTS "${CYRUS_SIGNED_FIXTURES}/lab_public_key.h")
    message(FATAL_ERROR "Local client requires its own ephemeral issuer public key")
  endif()
  set(CYRUS_BUILD_TOKEN_PROTOTYPE ON CACHE BOOL "Windows signed-token experiment" FORCE)
endif()
if(CYRUS_BUILD_SIGNED_LICENSE_LAB)
  if(NOT EXISTS "${CYRUS_SIGNED_FIXTURES}/lab_public_key.h" OR NOT EXISTS "${CYRUS_SIGNED_FIXTURES}/vectors.json")
    message(FATAL_ERROR "Generate ephemeral signed fixtures and set CYRUS_SIGNED_FIXTURES")
  endif()
  set(CYRUS_BUILD_TOKEN_PROTOTYPE ON CACHE BOOL "Windows signed-token experiment" FORCE)
endif()
enable_testing()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../CyrusLicensing" policy)
add_library(CyrusLicenseLab MODULE bridge.cpp)
target_compile_definitions(CyrusLicenseLab PRIVATE CYRUS_LICENSE_LAB_ONLY=1)
target_link_libraries(CyrusLicenseLab PRIVATE cyrus_license_policy core geom maxutil Maxscrpt)
cyrus_max_target(CyrusLicenseLab)
set_target_properties(CyrusLicenseLab PROPERTIES SUFFIX ".dlx")
if(CYRUS_BUILD_LOCAL_CLIENT_LAB)
  add_executable(CyrusLocalClientLab local_client.cpp)
  target_compile_definitions(CyrusLocalClientLab PRIVATE CYRUS_SIGNED_LICENSE_LAB_ONLY=1 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(CyrusLocalClientLab PRIVATE "${CYRUS_SIGNED_FIXTURES}")
  target_include_directories(CyrusLocalClientLab SYSTEM PRIVATE "../../CyrusLicensing/third_party")
  target_compile_options(CyrusLocalClientLab PRIVATE /W4 /WX /permissive-)
  target_link_libraries(CyrusLocalClientLab PRIVATE cyrus_license_local ncrypt)
endif()
if(CYRUS_BUILD_SIGNED_LICENSE_LAB)
  add_library(CyrusSignedLicenseLab MODULE signed_bridge.cpp)
  target_compile_definitions(CyrusSignedLicenseLab PRIVATE CYRUS_SIGNED_LICENSE_LAB_ONLY=1)
  target_include_directories(CyrusSignedLicenseLab PRIVATE "${CYRUS_SIGNED_FIXTURES}")
  target_link_libraries(CyrusSignedLicenseLab PRIVATE cyrus_license_token core geom maxutil Maxscrpt)
  cyrus_max_target(CyrusSignedLicenseLab)
  set_target_properties(CyrusSignedLicenseLab PROPERTIES SUFFIX ".dlx")
  add_executable(signed_license_tests "../../CyrusLicensing/tests/signed_license_tests.cpp")
  target_include_directories(signed_license_tests PRIVATE "${CYRUS_SIGNED_FIXTURES}" "${CMAKE_CURRENT_LIST_DIR}")
  target_include_directories(signed_license_tests SYSTEM PRIVATE "../../CyrusLicensing/third_party")
  target_compile_definitions(signed_license_tests PRIVATE CYRUS_SIGNED_LICENSE_LAB_ONLY=1)
  target_compile_options(signed_license_tests PRIVATE /W4 /WX /permissive-)
  target_link_libraries(signed_license_tests PRIVATE cyrus_license_token)
  add_executable(license_permit_benchmark permit_benchmark.cpp)
  target_compile_definitions(license_permit_benchmark PRIVATE CYRUS_SIGNED_LICENSE_LAB_ONLY=1)
  target_include_directories(license_permit_benchmark PRIVATE "${CYRUS_SIGNED_FIXTURES}")
  target_include_directories(license_permit_benchmark SYSTEM PRIVATE "../../CyrusLicensing/third_party")
  target_link_libraries(license_permit_benchmark PRIVATE cyrus_license_token)
  foreach(group profile signature claims scope lifecycle isolation permits publication transfer activation)
    add_test(NAME signed_license_${group} COMMAND signed_license_tests "${CYRUS_SIGNED_FIXTURES}/vectors.json" ${group})
  endforeach()
endif()
