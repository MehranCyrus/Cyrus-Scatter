cmake_minimum_required(VERSION 3.24)
project(CyrusVegetationLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
option(CVL_BUILD_MAX "Build the Max 2027 test adapter" ON)
include(CTest)
add_library(cyrus_prototype_contract INTERFACE)
target_include_directories(cyrus_prototype_contract INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/core/include")
target_compile_features(cyrus_prototype_contract INTERFACE cxx_std_20)
add_subdirectory(core/collision)
add_subdirectory(core/fitter)
add_subdirectory(core/delivery)
file(GLOB_RECURSE CVL_ID_INPUTS CONFIGURE_DEPENDS "core/*.hpp" "core/*.cpp" "core/*.cmake" "core/*.sha256" "include/*.hpp" "src/*.cpp" "src/*.hpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CVL_ID_INPUTS})
set(CVL_SOURCE_HASHES "lab-0.1.0")
foreach(input IN LISTS CVL_ID_INPUTS)
    file(SHA256 "${input}" part)
    string(APPEND CVL_SOURCE_HASHES "${part}")
endforeach()
string(SHA256 CVL_CORE_ID "${CVL_SOURCE_HASHES}")
add_library(cvl_kernel STATIC src/lab_core.cpp)
target_include_directories(cvl_kernel PUBLIC include "${CMAKE_CURRENT_SOURCE_DIR}/../CyrusLicensing/third_party")
target_link_libraries(cvl_kernel PUBLIC cyrus_vegetation_collision cyrus_vegetation_fitter cyrus_vegetation_delivery)
target_compile_definitions(cvl_kernel PUBLIC NOMINMAX CVL_CORE_ID="${CVL_CORE_ID}")
if(MSVC)
    target_compile_options(cvl_kernel PRIVATE /W4 /EHsc /permissive- /fp:precise)
endif()
if(BUILD_TESTING)
    add_executable(cvl_kernel_tests tests/kernel_tests.cpp)
    target_link_libraries(cvl_kernel_tests PRIVATE cvl_kernel)
    add_test(NAME vegetation_lab_kernel COMMAND cvl_kernel_tests)
    set_tests_properties(vegetation_lab_kernel PROPERTIES TIMEOUT 60)
endif()
if(CVL_BUILD_MAX)
    set(CYRUS_MAX_YEAR "2027" CACHE STRING "Max host year; this lab supports 2027 only")
    if(NOT CYRUS_MAX_YEAR EQUAL 2027)
        message(FATAL_ERROR "Cyrus Vegetation Lab 0.1.0 supports Max 2027 only")
    endif()
    include("${CMAKE_CURRENT_LIST_DIR}/../cmake/CyrusMaxSDK.cmake")
    add_library(CyrusVegetationLab SHARED src/max_host.cpp)
    cyrus_max_target(CyrusVegetationLab)
    target_link_libraries(CyrusVegetationLab PRIVATE cvl_kernel core geom mesh gfx maxutil Maxscrpt paramblk2)
    target_compile_options(CyrusVegetationLab PRIVATE /W4 /EHsc /permissive- /fp:precise)
    set_target_properties(CyrusVegetationLab PROPERTIES SUFFIX ".dlx")
    add_library(CyrusVegetationStorage MODULE src/storage_plugin.cpp)
    cyrus_max_target(CyrusVegetationStorage)
    target_link_libraries(CyrusVegetationStorage PRIVATE CyrusVegetationLab)
    set_target_properties(CyrusVegetationStorage PROPERTIES SUFFIX ".dlh")
endif()
