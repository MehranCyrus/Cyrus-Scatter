cmake_minimum_required(VERSION 3.24)
project(CyrusVegetationStemLab VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CYRUS_MAX_YEAR 2027 CACHE STRING "Host year")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/CyrusMaxSDK.cmake")
if(NOT CYRUS_MAX_YEAR EQUAL 2027)
    message(FATAL_ERROR "Stem Lab requires Max 2027")
endif()
add_library(CyrusStemInspect SHARED inspect.cpp)
cyrus_max_target(CyrusStemInspect)
target_include_directories(CyrusStemInspect PRIVATE "../../../CyrusLicensing/third_party")
target_link_libraries(CyrusStemInspect PRIVATE core geom mesh gfx maxutil Maxscrpt paramblk2)
target_compile_options(CyrusStemInspect PRIVATE /W4 /EHsc /permissive- /fp:precise)
set_target_properties(CyrusStemInspect PROPERTIES SUFFIX ".dlx")
set(LAB "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(CORE_REUSE "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/vegetation-dense-probe-20261024-06")
add_library(Jolt STATIC IMPORTED)
set_target_properties(Jolt PROPERTIES IMPORTED_LOCATION "${CORE_REUSE}/collision/jolt/Jolt.lib")
add_library(cyrus_collision STATIC IMPORTED)
set_target_properties(cyrus_collision PROPERTIES IMPORTED_LOCATION "${CORE_REUSE}/collision/cyrus_vegetation_collision.lib" INTERFACE_LINK_LIBRARIES Jolt)
add_library(cyrus_fitter STATIC IMPORTED)
set_target_properties(cyrus_fitter PROPERTIES IMPORTED_LOCATION "${CORE_REUSE}/fitter/cyrus_vegetation_fitter.lib")
add_library(stem_core STATIC stem_core.cpp analysis.cpp connectivity.cpp)
target_include_directories(stem_core PUBLIC "${LAB}/core/include" "${LAB}/core/fitter/private" "${LAB}/core/delivery/src" "../../../CyrusLicensing/third_party")
target_compile_options(stem_core PRIVATE /W4 /EHsc /permissive- /fp:precise)
target_compile_definitions(stem_core PRIVATE NOMINMAX)
target_link_libraries(stem_core PUBLIC cyrus_collision cyrus_fitter bcrypt)
add_executable(stem_kernel_test kernel_test.cpp)
target_link_libraries(stem_kernel_test PRIVATE stem_core)
add_executable(preparation_test analysis_test.cpp)
target_link_libraries(preparation_test PRIVATE stem_core)
add_executable(connectivity_test connectivity_test.cpp)
target_link_libraries(connectivity_test PRIVATE stem_core)
add_executable(asset_fit_test asset_fit_test.cpp)
target_link_libraries(asset_fit_test PRIVATE stem_core)
add_library(CyrusVegetationStemLab SHARED plugin.cpp)
cyrus_max_target(CyrusVegetationStemLab)
target_link_libraries(CyrusVegetationStemLab PRIVATE stem_core core geom mesh gfx maxutil Maxscrpt paramblk2 bcrypt)
target_compile_options(CyrusVegetationStemLab PRIVATE /W4 /EHsc /permissive- /fp:precise)
set_target_properties(CyrusVegetationStemLab PROPERTIES SUFFIX ".dlx")
add_library(CyrusVegetationStemGeometry MODULE storage.cpp)
cyrus_max_target(CyrusVegetationStemGeometry)
target_link_libraries(CyrusVegetationStemGeometry PRIVATE CyrusVegetationStemLab)
set_target_properties(CyrusVegetationStemGeometry PROPERTIES SUFFIX ".dlo")
