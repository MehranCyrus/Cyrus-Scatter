cmake_minimum_required(VERSION 3.24)
project(CyrusPhysicsLab VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
option(CPL_BUILD_MAX "Build independent Max 2027 modules" ON)
option(CPL_TEST_FAULTS "Private test injection; never package" OFF)
set(JOLT_SOURCE "" CACHE PATH "Pinned Jolt checkout")
set(PHYSX_SOURCE "" CACHE PATH "Pinned PhysX checkout")
foreach(flag TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER
    GENERATE_DEBUG_SYMBOLS INTERPROCEDURAL_OPTIMIZATION OVERRIDE_CXX_FLAGS ENABLE_ALL_WARNINGS
    FLOATING_POINT_EXCEPTIONS_ENABLED USE_STATIC_MSVC_RUNTIME_LIBRARY DEBUG_RENDERER_IN_DEBUG_AND_RELEASE
    PROFILER_IN_DEBUG_AND_RELEASE ENABLE_OBJECT_STREAM ENABLE_INSTALL JPH_USE_DX12 JPH_USE_VK JPH_USE_MTL
    USE_AVX USE_AVX2 USE_LZCNT USE_TZCNT USE_F16C USE_FMADD USE_SSE4_1 USE_SSE4_2)
  set(${flag} OFF CACHE BOOL "Laboratory build policy" FORCE)
endforeach()
set(CPP_EXCEPTIONS_ENABLED ON CACHE BOOL "" FORCE)
set(CPP_RTTI_ENABLED ON CACHE BOOL "" FORCE)
add_subdirectory("${JOLT_SOURCE}/Build" "${CMAKE_BINARY_DIR}/jolt" EXCLUDE_FROM_ALL)
foreach(flag PX_GENERATE_GPU_PROJECTS PX_BUILDSNIPPETS PX_BUILDPVDRUNTIME NV_USE_STATIC_WINCRT)
  set(${flag} OFF CACHE BOOL "CPU comparison only" FORCE)
endforeach()
set(PX_GENERATE_STATIC_LIBRARIES ON CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 14)
add_subdirectory("${PHYSX_SOURCE}/physx" "${CMAKE_BINARY_DIR}/physx" EXCLUDE_FROM_ALL)
set(CMAKE_CXX_STANDARD 20)
add_library(cpl_core STATIC src/core.cpp src/jolt_world.cpp src/live_session.cpp src/quality.cpp)
set(CPL_SOLVER_SOURCES "${CMAKE_CURRENT_LIST_FILE}" include/core.h include/live_session.h src/core.cpp src/jolt_world.cpp src/jolt_collider.inc src/live_session.cpp src/quality.cpp third_party/vhacd/VHACD.h)
set(CPL_SOLVER_HASHES "jolt-1b2e77e3c6c09709e5edc79dc208459b8ff32051")
foreach(source IN LISTS CPL_SOLVER_SOURCES)
  if(NOT IS_ABSOLUTE "${source}")
    set(source "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  endif()
  file(SHA256 "${source}" hash)
  string(APPEND CPL_SOLVER_HASHES "${hash}")
endforeach()
string(SHA256 CPL_SOLVER_ID "${CPL_SOLVER_HASHES}")
target_compile_definitions(cpl_core PUBLIC CPL_SOLVER_ID="${CPL_SOLVER_ID}")
target_include_directories(cpl_core PUBLIC include ../CyrusLicensing/third_party)
target_link_libraries(cpl_core PUBLIC Jolt)
target_compile_definitions(cpl_core PUBLIC NOMINMAX)
if(MSVC)
  target_compile_options(cpl_core PRIVATE /W4 /EHsc /permissive-)
endif()
add_library(cpl_physx STATIC src/physx_world.cpp)
target_compile_definitions(cpl_physx PRIVATE PX_PHYSX_STATIC_LIB)
target_include_directories(cpl_physx PUBLIC include "${PHYSX_SOURCE}/physx/include")
target_link_libraries(cpl_physx PUBLIC cpl_core PhysX PhysXCommon PhysXFoundation PhysXExtensions)
add_executable(cpl_compare tests/compare.cpp)
target_link_libraries(cpl_compare PRIVATE cpl_physx psapi)
add_executable(cpl_tests tests/core_tests.cpp)
target_compile_definitions(cpl_tests PRIVATE CPL_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
target_link_libraries(cpl_tests PRIVATE cpl_physx)
enable_testing()
add_executable(cpl_hull_tests tests/hull_tests.cpp)
target_link_libraries(cpl_hull_tests PRIVATE cpl_core)
target_compile_definitions(cpl_hull_tests PRIVATE CPL_HULL_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME physics_artist_hulls COMMAND cpl_hull_tests "${CMAKE_BINARY_DIR}/hull-results.json")
set_tests_properties(physics_artist_hulls PROPERTIES TIMEOUT 120)
add_executable(cpl_quality_tests tests/quality_tests.cpp)
target_link_libraries(cpl_quality_tests PRIVATE cpl_core)
target_compile_definitions(cpl_quality_tests PRIVATE CPL_QUALITY_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME physics_collision_quality COMMAND cpl_quality_tests "${CMAKE_BINARY_DIR}/quality-results.json")
set_tests_properties(physics_collision_quality PROPERTIES TIMEOUT 120)
add_test(NAME physics_core COMMAND cpl_tests)
set_tests_properties(physics_core PROPERTIES TIMEOUT 30)
add_executable(cpl_live_tests tests/live_tests.cpp)
target_link_libraries(cpl_live_tests PRIVATE cpl_core)
add_test(NAME physics_continuous COMMAND cpl_live_tests)
set_tests_properties(physics_continuous PROPERTIES TIMEOUT 60)
add_executable(cpl_live_stress tests/live_stress.cpp)
target_link_libraries(cpl_live_stress PRIVATE cpl_core psapi)
add_executable(cpl_grip_tests tests/grip_tests.cpp)
target_link_libraries(cpl_grip_tests PRIVATE cpl_core)
add_test(NAME physics_grip COMMAND cpl_grip_tests "${CMAKE_BINARY_DIR}/grip-results.json")
set_tests_properties(physics_grip PROPERTIES TIMEOUT 120)
add_executable(cpl_collision_tests tests/collision_tests.cpp)
target_link_libraries(cpl_collision_tests PRIVATE cpl_core)
add_test(NAME physics_collision_bindings COMMAND cpl_collision_tests "${CMAKE_BINARY_DIR}/collision-results.json")
set_tests_properties(physics_collision_bindings PROPERTIES TIMEOUT 60)
add_executable(cpl_mountain_tests tests/mountain_tests.cpp)
target_link_libraries(cpl_mountain_tests PRIVATE cpl_core)
add_test(NAME physics_mountain_display COMMAND cpl_mountain_tests "${CMAKE_BINARY_DIR}/mountain-results.json")
set_tests_properties(physics_mountain_display PROPERTIES TIMEOUT 60)
add_executable(cpl_collider_cook_tests tests/collider_cook_tests.cpp)
target_link_libraries(cpl_collider_cook_tests PRIVATE cpl_core)
target_compile_definitions(cpl_collider_cook_tests PRIVATE CPL_COOK_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME physics_small_baked_colliders COMMAND cpl_collider_cook_tests "${CMAKE_BINARY_DIR}/collider-cook-results.json")
set_tests_properties(physics_small_baked_colliders PROPERTIES TIMEOUT 120)
if(CPL_BUILD_MAX)
  if(NOT CYRUS_MAX_YEAR EQUAL 2027)
    message(FATAL_ERROR "This standalone laboratory supports Max 2027 only")
  endif()
  include("${CMAKE_CURRENT_LIST_DIR}/../cmake/CyrusMaxSDK.cmake")
  add_library(CyrusPhysicsLab SHARED src/max_host.cpp src/diagnostics.cpp)
  target_link_libraries(CyrusPhysicsLab PRIVATE cpl_core core geom mesh gfx maxutil Maxscrpt paramblk2 psapi Poly MNMath)
  cyrus_max_target(CyrusPhysicsLab)
  target_compile_options(CyrusPhysicsLab PRIVATE /Zi)
  target_link_options(CyrusPhysicsLab PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF)
  add_executable(CyrusPhysicsCrashWatch src/crash_watch.cpp)
  target_include_directories(CyrusPhysicsCrashWatch PRIVATE include)
  target_compile_definitions(CyrusPhysicsCrashWatch PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(CyrusPhysicsCrashWatch PRIVATE dbghelp)
  add_dependencies(CyrusPhysicsLab CyrusPhysicsCrashWatch)
  if(CPL_TEST_FAULTS)
    target_compile_definitions(CyrusPhysicsLab PRIVATE CPL_TEST_FAULTS=1)
  endif()
  set_target_properties(CyrusPhysicsLab PROPERTIES SUFFIX ".dlx")
  add_library(CyrusPhysicsStorage MODULE src/storage_plugin.cpp)
  add_dependencies(CyrusPhysicsStorage CyrusPhysicsLab)
  target_link_libraries(CyrusPhysicsStorage PRIVATE CyrusPhysicsLab)
  cyrus_max_target(CyrusPhysicsStorage)
  set_target_properties(CyrusPhysicsStorage PROPERTIES SUFFIX ".dlh")
endif()
