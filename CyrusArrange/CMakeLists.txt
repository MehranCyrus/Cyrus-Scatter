cmake_minimum_required(VERSION 3.24)
project(CyrusArrange VERSION 0.3.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CYRUS_ARRANGE_BUILD_MAX "Build the 3ds Max 2027 extension" ON)
add_library(arrange_clipper STATIC third_party/clipper2/src/clipper.engine.cpp third_party/clipper2/src/clipper.offset.cpp third_party/clipper2/src/clipper.rectclip.cpp)
target_include_directories(arrange_clipper PUBLIC third_party/clipper2/include)
add_library(arrange_geometry STATIC src/geometry.cpp src/receiving_mesh.cpp src/assets.cpp src/arrangement.cpp src/placement_queries.cpp src/region_registry.cpp src/document_state.cpp src/geometry_cache.cpp src/path.cpp src/collision_proxy.cpp src/fit_feedback.cpp src/contact_base.cpp)
target_include_directories(arrange_geometry PUBLIC include)
target_link_libraries(arrange_geometry PRIVATE arrange_clipper)
if(MSVC)
  target_compile_options(arrange_geometry PRIVATE /W4 /permissive- /EHsc /Zi)
endif()
enable_testing()
add_executable(arrange_geometry_tests tests/geometry_tests.cpp)
target_link_libraries(arrange_geometry_tests PRIVATE arrange_geometry)
add_test(NAME arrange_geometry COMMAND arrange_geometry_tests)
foreach(suite receiving_mesh assets arrangement placement_queries integrated_arrangement region_registry composition composition_golden document_state query_accounting geometry_cache numeric_support space_workflow path path_state packing packing_state collision_proxy fit_feedback contact_base)
  add_executable(arrange_${suite}_tests tests/${suite}_tests.cpp)
  target_link_libraries(arrange_${suite}_tests PRIVATE arrange_geometry)
  add_test(NAME arrange_${suite} COMMAND arrange_${suite}_tests)
endforeach()
target_include_directories(arrange_query_accounting_tests PRIVATE src third_party/clipper2/include)
if(CYRUS_ARRANGE_BUILD_MAX)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Cyrus Arrange requires Windows x64 for Max")
  endif()
  set(MAXSDK_ROOT "C:/Program Files/Autodesk/3ds Max 2027 SDK/maxsdk" CACHE PATH "3ds Max 2027 SDK root")
  if(NOT EXISTS "${MAXSDK_ROOT}/include/maxversion.h")
    message(FATAL_ERROR "Set MAXSDK_ROOT to the Max 2027 maxsdk folder")
  endif()
  file(STRINGS "${MAXSDK_ROOT}/include/maxversion.h" _sdk_year REGEX "^#define MAX_PRODUCT_YEAR_NUMBER +[0-9]+")
  if(NOT _sdk_year MATCHES "MAX_PRODUCT_YEAR_NUMBER +2027($|[^0-9])")
    message(FATAL_ERROR "Only the Max 2027 SDK is supported")
  endif()
  enable_language(RC)
  add_library(CyrusArrange SHARED src/max_bridge.cpp src/arrange_bridge.cpp src/arrange_document.cpp src/preview_service.cpp src/path_capture.cpp src/arrange_presentation.cpp src/arrange_presentation.rc)
  target_include_directories(CyrusArrange SYSTEM PRIVATE "${MAXSDK_ROOT}/include")
  target_link_directories(CyrusArrange PRIVATE "${MAXSDK_ROOT}/lib/x64/Release")
  target_link_libraries(CyrusArrange PRIVATE arrange_geometry core geom mesh gfx maxutil Maxscrpt paramblk2 ole32)
  target_compile_definitions(CyrusArrange PRIVATE UNICODE _UNICODE NOMINMAX CYRUS_MAX_YEAR=2027)
  target_compile_options(CyrusArrange PRIVATE /EHsc /utf-8 /Zi)
  target_link_options(CyrusArrange PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:REF /OPT:ICF)
  set_target_properties(CyrusArrange PROPERTIES PREFIX "" SUFFIX ".dlx")
  add_library(CyrusArrangeStorage MODULE src/storage_plugin.cpp)
  target_include_directories(CyrusArrangeStorage SYSTEM PRIVATE "${MAXSDK_ROOT}/include")
  target_link_directories(CyrusArrangeStorage PRIVATE "${MAXSDK_ROOT}/lib/x64/Release")
  target_link_libraries(CyrusArrangeStorage PRIVATE CyrusArrange core maxutil)
  target_compile_definitions(CyrusArrangeStorage PRIVATE UNICODE _UNICODE NOMINMAX)
  target_compile_options(CyrusArrangeStorage PRIVATE /EHsc /utf-8)
  set_target_properties(CyrusArrangeStorage PROPERTIES PREFIX "" SUFFIX ".dlh")
endif()
