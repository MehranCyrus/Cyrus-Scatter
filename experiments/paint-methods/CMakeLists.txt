cmake_minimum_required(VERSION 3.24)
project(PaintMethodsLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(PAINTLAB_MAX "Build independent Max 2027 modules" ON)
add_library(paintlab STATIC src/lab.cpp
 methods/01-vector-regions/src/vector.cpp methods/02-tiled-mask/src/mask.cpp
 methods/03-surface-volumes/src/volumes.cpp methods/04-surface-field/src/field.cpp
 third_party/clipper2/src/clipper.engine.cpp third_party/clipper2/src/clipper.offset.cpp third_party/clipper2/src/clipper.rectclip.cpp)
target_include_directories(paintlab PUBLIC src third_party/clipper2/include)
if(MSVC)
 target_compile_options(paintlab PRIVATE /W4 /O2)
endif()
enable_testing()
add_executable(lab_tests tests/core_tests.cpp)
target_link_libraries(lab_tests PRIVATE paintlab)
add_test(NAME paintlab_core COMMAND lab_tests)
add_executable(lab_benchmark tests/benchmark.cpp)
target_link_libraries(lab_benchmark PRIVATE paintlab)
add_executable(lab_diagnostics tests/diagnostics.cpp)
target_link_libraries(lab_diagnostics PRIVATE paintlab)
if(PAINTLAB_MAX)
 if(NOT EXISTS "${MAXSDK_ROOT}/include/max.h")
  message(FATAL_ERROR "Set MAXSDK_ROOT to Max 2027 SDK")
 endif()
 foreach(method RANGE 1 4)
  add_library(PaintLab${method} MODULE src/max_host.cpp)
  target_include_directories(PaintLab${method} SYSTEM PRIVATE "${MAXSDK_ROOT}/include")
  target_link_directories(PaintLab${method} PRIVATE "${MAXSDK_ROOT}/lib/x64/Release")
  target_link_libraries(PaintLab${method} PRIVATE paintlab core geom mesh gfx maxutil Maxscrpt paramblk2)
  target_compile_definitions(PaintLab${method} PRIVATE UNICODE _UNICODE NOMINMAX LAB_METHOD=${method})
  set_target_properties(PaintLab${method} PROPERTIES SUFFIX ".dlx")
 endforeach()
endif()
