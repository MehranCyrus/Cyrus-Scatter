cmake_minimum_required(VERSION 3.24)
project(CyrusLicensingPolicy VERSION 0.1.0 LANGUAGES CXX)

# Isolated policy by default. The candidate Windows verifier is opt-in and is
# not linked to product targets. Issuer fixtures belong only to lab targets.
add_library(cyrus_license_policy STATIC src/policy.cpp)
target_include_directories(cyrus_license_policy PUBLIC include)
target_compile_features(cyrus_license_policy PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(cyrus_license_policy PRIVATE /W4 /WX /permissive-)
endif()

option(CYRUS_BUILD_TOKEN_PROTOTYPE "Build isolated Windows signed-license verifier" OFF)
if(CYRUS_BUILD_TOKEN_PROTOTYPE)
  if(NOT WIN32)
    message(FATAL_ERROR "The candidate verifier uses Windows CNG")
  endif()
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann/json.hpp" _json_hash)
  if(NOT _json_hash STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
    message(FATAL_ERROR "Pinned nlohmann/json 3.12.0 header fingerprint differs")
  endif()
  add_library(cyrus_license_token STATIC src/signed_license.cpp src/cng_signature.cpp src/operation_permit.cpp)
  target_include_directories(cyrus_license_token PUBLIC include)
  target_include_directories(cyrus_license_token SYSTEM PRIVATE third_party)
  target_compile_features(cyrus_license_token PUBLIC cxx_std_20)
  target_compile_definitions(cyrus_license_token PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(cyrus_license_token PUBLIC cyrus_license_policy PRIVATE bcrypt crypt32)
  if(MSVC)
    target_compile_options(cyrus_license_token PRIVATE /W4 /WX /permissive-)
  endif()
  add_library(cyrus_license_local STATIC src/local_clock.cpp src/local_store.cpp src/installation_key.cpp src/client.cpp)
  target_include_directories(cyrus_license_local PUBLIC include)
  target_include_directories(cyrus_license_local SYSTEM PRIVATE third_party)
  target_compile_features(cyrus_license_local PUBLIC cxx_std_20)
  target_compile_definitions(cyrus_license_local PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(cyrus_license_local PUBLIC cyrus_license_token PRIVATE ncrypt bcrypt crypt32)
  if(MSVC)
    target_compile_options(cyrus_license_local PRIVATE /W4 /WX /permissive-)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  if(TARGET cyrus_license_local)
    add_executable(license_local_tests tests/local_state_tests.cpp)
    target_link_libraries(license_local_tests PRIVATE cyrus_license_local ncrypt)
    if(MSVC)
      target_compile_options(license_local_tests PRIVATE /W4 /WX /permissive-)
    endif()
    foreach(group clock key store)
      add_test(NAME license_local_${group} COMMAND license_local_tests ${group} "${CMAKE_BINARY_DIR}/local-fixtures")
    endforeach()
  endif()
  add_executable(license_policy_tests tests/policy_tests.cpp)
  target_link_libraries(license_policy_tests PRIVATE cyrus_license_policy)
  if(MSVC)
    target_compile_options(license_policy_tests PRIVATE /W4 /WX /permissive-)
  endif()
  foreach(group defaults lifecycle continuity facts deadlines extension)
    add_test(NAME license_policy_${group} COMMAND license_policy_tests ${group})
  endforeach()
endif()
